# Python 3.12.
#
# The versions below are pinned because this repository claims exact numbers:
# every figure caption and every value in the module READMEs is printed by the
# notebooks, and both the RandomForest and LIME's own sampling are sensitive to
# the RNG implementation. A different scikit-learn or numpy will still run, and
# the conclusions will hold, but the third decimal will move and the committed
# figures will no longer match byte for byte.
#
# Relax the pins if you only want to follow along; keep them if you want to
# reproduce the committed outputs.

lime==0.2.0.1
numpy==2.4.6
scipy==1.18.0
matplotlib==3.11.1

# The SRAG modules read the Ministry's parquet microdata. Pinned for the same
# reason as the rest: the profile and quality reports are committed artefacts,
# and a pandas major changes the default string dtype, hence the parquet bytes.
pandas==2.3.3
pyarrow==23.0.1

# tools/ only. pypdf parses the official data dictionary; duckdb loads the
# parquet into the local Postgres. Neither touches a printed number.
pypdf==6.19.0
duckdb==1.4.4

# Tooling. Not pinned — it does not affect any result.
# jupytext is optional — handy for diffing notebooks as text; no .py sources
# are tracked.
jupyter
jupytext
